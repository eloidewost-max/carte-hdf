359:   function computeICPScore(code, returnDetail) {
360:     if (!returnDetail && icpScoreCache && icpScoreCache[code] !== undefined) {
361:       return icpScoreCache[code];
362:     }
363:     var d = hdf[code];
364:     if (!d) return null;
365:     var pop = d.pop || 0;
366: 
367:     var signals = {};
368: 
369:     // pm_count: agents PM + ASVP per 10k pop, cap 10
370:     var pmTotal = (d.pm || 0) + (d.asvp || 0);
371:     signals.pm_count = pop > 0 ? Math.min(pmTotal / pop * 10000 / 10, 1) : 0;
372: 
373:     // inondable: binary from hardcoded DREAL list
374:     signals.inondable = INOND_SET[code] ? 1 : 0;
375: 
376:     // crime_r: taux délinquance /10k (delinquance.json), cap 500/10k
377:     signals.crime_r = (d.crime_r != null) ? Math.min(d.crime_r / 500, 1) : 0;
378: 
379:     // tourisme: binary from hardcoded list
380:     signals.tourisme = TOUR_SET[code] ? 1 : 0;
381: 
382:     var totalWeight = 0, weightedSum = 0;
383:     for (var k in signals) {
384:       var w = icpWeights[k] || 0;
385:       totalWeight += w;
386:       weightedSum += signals[k] * w;
387:     }
388:     if (totalWeight === 0) return null;
389:     var score = Math.round(weightedSum / totalWeight * 100);
390: 
391:     if (returnDetail) {
392:       var detail = [];
393:       ICP_SIGNALS.forEach(function(sig) {
394:         var w = icpWeights[sig.key] || 0;
395:         var raw = signals[sig.key] || 0;
396:         var contrib = totalWeight > 0 ? Math.round(raw * w / totalWeight * 100) : 0;
397:         detail.push({ key: sig.key, label: sig.label, raw: raw, weight: w, contrib: contrib, desc: sig.desc });
398:       });
399:       return { score: score, detail: detail };
400:     }
401: 
402:     if (!icpScoreCache) icpScoreCache = {};
403:     icpScoreCache[code] = score;
404:     return score;
405:   }

1004:   function decodePolyline(str) {
1005:     var index = 0, lat = 0, lng = 0, coordinates = [];
1006:     while (index < str.length) {
1007:       var b, shift = 0, result = 0;
1008:       do { b = str.charCodeAt(index++) - 63; result |= (b & 0x1f) << shift; shift += 5; } while (b >= 0x20);
1009:       var dlat = (result & 1) ? ~(result >> 1) : (result >> 1); lat += dlat;
1010:       shift = 0; result = 0;
1011:       do { b = str.charCodeAt(index++) - 63; result |= (b & 0x1f) << shift; shift += 5; } while (b >= 0x20);
1012:       var dlng = (result & 1) ? ~(result >> 1) : (result >> 1); lng += dlng;
1013:       coordinates.push([lat / 1e5, lng / 1e5]);
1014:     }
1015:     return coordinates;
1016:   }
